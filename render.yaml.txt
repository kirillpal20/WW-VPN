services:
  # 1. Бот + выдача подписки
  - type: web
    name: vpn-sub-bot
    runtime: python
    plan: free
    buildCommand: "bash build.sh"
    startCommand: "XRAY_BIN=./xray uvicorn bot_server:app --host 0.0.0.0 --port $PORT"
    envVars:
      - key: BOT_TOKEN
        sync: false          # впишешь вручную в панели Render (секрет)
      - key: BASE_URL
        sync: false          # впишешь после первого деплоя, когда узнаешь адрес сервиса
      - key: WEBHOOK_SECRET
        generateValue: true  # Render сам сгенерирует случайную строку
      - key: REDIS_URL
        fromService:
          name: vpn-sub-redis
          type: keyvalue
          property: connectionString

  # 2. Проверка серверов раз в час
  - type: cron
    name: vpn-sub-checker
    runtime: python
    plan: free
    schedule: "0 * * * *"     # каждый час в 00 минут
    buildCommand: "bash build.sh"
    startCommand: "XRAY_BIN=./xray python checker.py"
    envVars:
      - key: REDIS_URL
        fromService:
          name: vpn-sub-redis
          type: keyvalue
          property: connectionString

  # 3. Хранилище (общая база для бота и чекера)
  - type: keyvalue
    name: vpn-sub-redis
    plan: free
    ipAllowList: []
